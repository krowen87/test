# GIT

git init // добавление локального хранилища в git

 git remote -v  // проверка привязки репозитория

 git remote add origin (ссылка на папку удалённого репозитория) // связь локального и удаленного репозитория

 git push -u origin main // отправка данных в удаленный репозиторий (имя пишем при первой отправке)

  git clone (ссылка на форк репозитория) // добавление скопированного репозитория на локальное хранилище

  cd (имя папки) // переход между репозиториями в терминале

  git branch // работа с ветками репозитория

  git branch (имя) // создание новой ветки

  git checkout (имя ветки) // переключение между ветками

  git checkout -b // переключение с созданием ветки

   git status // проверка состояния, изменения

   git add (имя) // добавление файла в отслеживание
   
   git add -A // добавление всех файлов в отслеживаемые

   git commit -m "Название коммита" // создание коммита (add-commit-push)

   git log (имя-ветки) --graph --oneline // просмотр истории ветки

   git merge имя ветки // слияние веток - находимся в ветке, в которую слияние, а пишем в команде ветку, которую обьединяем

   .gitignore — это файл-настройка, который позволяет игнорировать нужные папки и файлы во время публикации в удалённый репозиторий на GitHub

   Ключевые действия и команды при командной разработке

Для обновления командного проекта на GitHub нужно использовать команду ##git pull
Чтобы переключаться к предыдущему состоянию проекта, используйте команду ##git checkout хеш коммита
Посмотреть всю историю проекта можно через команду ##git log --oneline --all
Чтобы отменить коммит в удалённом репозитории, нужно использовать команду ##git revert хеш коммита для отмены
Настроить ##.gitignore можно для папки и файла с помощью команды ##git rm --cached имя-файла
